hollow-star-os-basic/                # Główny folder repozytorium (otwierasz go w Antigravity)
├── AGENTS.md                        # Stały kontekst projektu dla Google Antigravity
├── LICENSE                          # Licencja GNU General Public License v3.0
├── README.md                        # Dokumentacja projektu po angielsku (GitHub)
├── .gitignore                       # Ignoruje pliki tymczasowe kompilacji (zachowuje releases/*.uf2)
├── pico.sh                          # Skrypt: build [picow|pico|all] / release / upload / flash / monitor / logs
├── docs/
│   └── wiring.md                    # Schemat podłączenia pinów (ST7565, SD, USB 2.4 GHz, Audio Jack PWM)
├── releases/                        # Gotowe flashowalne obrazy .uf2 dla każdej wersji (Pico i Pico W)
│   ├── hollow-star-os-pico-v1.1.0.uf2
│   └── hollow-star-os-picow-v1.1.0.uf2
└── hollow_star_os_basic/            # Folder szkicu Arduino (otwierany też przez Arduino IDE)
    ├── hollow_star_os_basic.ino     # Wątek UI (Core 0), routing ekranów, F1 pomoc, suspend, Core 1 USB loop
    ├── usb_kbd.h / .cpp             # Klawiatura USB Host na PIO (Core 1, izolacja TinyUSB)
    ├── storage_cfg.h / .cpp         # LittleFS, karta SD (SPI1/SDFS), /config.cfg, wolne miejsce, formatowanie
    ├── bt_audio.h / .cpp            # Wątek Audio (A2DP + PWMAudio Jack + detekcja GP8 + libhelix + głośność)
    ├── net_ftp.h / .cpp             # Wątek sieciowy Pico W (Wi-Fi + RSSI, NTP, FTP z logowaniem)
    └── src/                         # Moduły UI, aplikacji i gier
        ├── os/
        │   ├── app_types.h          # Stany AppState, detekcja Pico/Pico W (HSOS_HAS_WIRELESS), modale, AppScreen
        │   ├── ui_common.h / .cpp   # Wspólny scratch RAM, pasek stanu (Wi-Fi/BT/Jack), branding, modale, pomoc F1
        │   └── ui_menus.h / .cpp    # Menu główne i podmenu (ukrywa BT/Wi-Fi/NTP/FTP na zwykłym Pico)
        ├── apps/
        │   ├── app_temp.h / .cpp    # Sys Info: temp. RP2040, zajęty RAM, wolne miejsce Flash i SD
        │   ├── app_calc.h / .cpp    # Kalkulator (ułamki, pierwiastki, potęgi, dłuższe wyrażenia)
        │   ├── app_notes.h / .cpp   # Kreator lokalizacji/nazwy + edytor/podgląd Markdown (.md)
        │   ├── app_files.h / .cpp   # Eksplorator plików (Flash / SD)
        │   ├── app_music.h / .cpp   # Odtwarzacz MP3 (Bluetooth A2DP / Headphone Jack PWM + głośność + equalizer)
        │   ├── app_ftp.h / .cpp     # Interfejs serwera FTP (dysk Flash/SD + konfiguracja loginu i hasła)
        │   └── app_settings.h / .cpp# Ustawienia (BT, Wi-Fi/NTP dla Pico W; Kontrast, Czas, Formatowanie, About)
        └── games/
            ├── game_brick.h / .cpp  # Brick Breaker
            ├── game_tetris.h / .cpp # Tetris
            └── game_racing.h / .cpp # Wyścigi (Brick Game)
